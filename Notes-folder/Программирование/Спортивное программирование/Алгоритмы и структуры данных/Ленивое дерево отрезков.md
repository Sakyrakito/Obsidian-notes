```c++
struct Node
{
    int sm, ln, add;
    
    Node(int v = 0)
        : sm(v), ln(1), add(0)
    {
    }
};

class LazySegmentTree
{
public:
    LazySegmentTree(int sz)
    {
        n = sz;
        tree.resize(n * 4);
    }
  
    void update(int l, int r, int val)
    {
        update_tree(l, r, val, 1, 1, n);
    }
  
    int get_sum(int l, int r){
        return get_sum_tree(l, r, 1, 1, n);
    }
  
private:
    int n;
    vector<Node> tree;
  
    void push(int v)
    {
        tree[v * 2].add += tree[v].add;
        tree[v * 2 + 1].add += tree[v].add;
        tree[v].add = 0;
    }
  
    Node combine(Node left, Node right){
        Node res;
  
        res.ln = left.ln + right.ln;
        res.sm = left.sm + left.add * left.ln + right.sm + right.add * right.ln;
  
        return res;
    }
  
    void update_tree(int l, int r, int val, int v, int tl, int tr)
    {
        if (tl >= l && tr <= r)
        {
            tree[v].add += val;
            return;
        }
  
        if (tl > r || tr < l)
        {
            return;
        }
  
        push(v);
        int mid = (tl + tr) / 2;
        update_tree(l, r, val, v * 2, tl, mid);
        update_tree(l, r, val, v * 2 + 1, mid + 1, tr);
  
        tree[v] = combine(tree[v * 2], tree[v * 2 + 1]);
    }
  
    int get_sum_tree(int l, int r, int v, int tl, int tr){
        if (tl > r || tr < l){
            return 0ll;
        }
  
        if (tl >= l && tr <= r){
            return tree[v].sm + tree[v].add * tree[v].ln;
        }
  
        push(v);
        int mid = (tl + tr) / 2;
        return get_sum_tree(l, r, v * 2, tl, mid) + get_sum_tree(l, r, v * 2 + 1, mid + 1, tr);
    }
};
```
