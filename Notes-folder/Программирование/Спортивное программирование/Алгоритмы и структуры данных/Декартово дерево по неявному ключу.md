>[!info]
>Тут реализованы только базовые операции insert, erase, get

```c++
class Treap {
private:

    struct Node {
        int value;
        int priority;
        int size;
        Node* l = nullptr;
        Node* r = nullptr;

        Node(int v)
            : value(v), priority(rnd()), size(1)
        {
        }
    };

    Node* root = nullptr;

    int get_size(Node* t) {
        return t ? (int)t->size : 0ll;
    }

    void update_size(Node* t) {
        if (t) {
            t->size = 1 + get_size(t->l) + get_size(t->r);
        }
    }

    void merge(Node*& t, Node* l, Node* r) {
        if (!l || !r) {
            t = l ? l : r;
        }
        else if (l->priority > r->priority) {
            merge(l->r, l->r, r);
            t = l;
        }
        else {
            merge(r->l, l, r->l);
            t = r;
        }

        update_size(t);
    }

    void split(Node* t, int k, Node*& l, Node*& r) {
        if (!t) {
            l = r = nullptr;
            return;
        }

        int left_size = get_size(t->l);
        if (left_size < k) {
            split(t->r, k - left_size - 1, t->r, r);
            l = t;
        }
        else {
            split(t->l, k, l, t->l);
            r = t;
        }

        update_size(t);
    }

public:
    void insert(int pos, int val) {
        Node* l, * r;
        split(root, pos, l, r);

        Node* newNode = new Node(val);

        merge(l, l, newNode);
        merge(root, l, r);
    }

    void erase(int pos) {
        Node* l, * m, * r;
        split(root, pos, l, m);
        split(m, 1, m, r);

        // delete m
        
        merge(root, l, r);
    }

    int get(int pos) {
        Node* l, * m, * r;
        split(root, pos, l, m);
        split(m, 1, m, r);

        int res = m->value;

        merge(m, m, r);
        merge(root, l, m);
        return res;
    }

    int size() {
        return get_size(root);
    }
};
```
