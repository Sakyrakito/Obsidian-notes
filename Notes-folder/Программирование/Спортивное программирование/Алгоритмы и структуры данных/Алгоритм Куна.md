>[!info]
>алгоритм для нахождения максимального паросочетания и минимального вершинного покрытия. Работает за O(N * M)

```c++
 vector<int> matchL(n), matchR(n);
 
 function<bool(int)> khun = [&](int u){
        for (auto v: g2[u]){
            if (!vis[v]){
                vis[v] = true;

                if (matchR[v] == 0 || khun(matchR[v])){
                    matchR[v] = u;
                    matchL[u] = v;

                    return true;
                }
            }
        }

        return false;
    };

    int M = 0;
    for (int i = 0; i < n; i++){
        vis.assign(n, false);

        if (khun(i)){
            M++;
        }
    }
    
// После запуска алгоритма Куна:
// matchL[u] = пара u, matchR[v] = пара v (0 = нет пары)

vector<bool> reachU(n+1, false), reachV(m+1, false);
queue<int> q;

// Стартуем из всех свободных вершин U
for (int u = 1; u <= n; u++)
    if (matchL[u] == 0) { q.push(u); reachU[u] = true; }

while (!q.empty()) {
    int u = q.front(); q.pop();
    for (int v : adj[u]) {
        if (reachV[v]) continue;
        reachV[v] = true;           // идём по ненасыщенному ребру в V
        int u2 = matchR[v];
        if (u2 && !reachU[u2]) {
            reachU[u2] = true;      // идём по насыщенному ребру обратно в U
            q.push(u2);
        }
    }
}

// Минимальное вершинное покрытие:
vector<int> cover;
for (int u = 1; u <= n; u++)
    if (!reachU[u]) cover.push_back(u);   // U \ reach
for (int v = 1; v <= m; v++)
    if (reachV[v])  cover.push_back(v);   // V ∩ reach (со смещением индексов)
```
