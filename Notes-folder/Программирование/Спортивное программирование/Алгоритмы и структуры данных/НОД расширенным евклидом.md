int gcd_ext(int a, int b, int& x, int& y) {
    if (b == 0) {
        x = 1; y = 0;
        return a;
    }
    int x1, y1;
    int d = gcd_ext(b, a % b, x1, y1);
    x = y1;
    y = x1 - (a / b) * y1;
    return d;
}
long long inv_euclid(long long a, long long m) {
    long long x, y;
    long long g = gcd_ext(a, m, x, y);
    return (x % m + m) % m;
}
